% Extra preamble after header.tex (which stays verbatim from the spike).
% Latin Modern Math has no U+00B5 MICRO SIGN, so xelatex silently dropped µ
% inside formulas. Typeset it as an upright mu in math; keep the text glyph
% (from the main font) in text. The formula source itself is not changed.
\usepackage{newunicodechar}
\newunicodechar{µ}{\ifmmode\upmu\else\symbol{"00B5}\fi}

% Missing glyphs are logged ("Missing character: There is no ...") so the
% service can report them; it never ships them silently.
\tracinglostchars=2

% Numbers exactly as written (owner decision): \num/\SI/\qty print their
% number without parsing or reformatting (1,500 / 12345 / .5 / 1.5e3 stay so).
% Unparsed, siunitx would print the input as plain math: a comma followed by
% punctuation space ("1, 500") and letters in italic (1.5𝑒3). So an unparsed
% number is printed upright (\symup), every comma as an ordinary character.
% This hooks siunitx's internal number printer: if siunitx renames it, the
% preamble stops with an error (and the typeset tests fail) instead of numbers
% silently changing. The web page's KaTeX macros print the same.
\sisetup{parse-numbers=false}
\ExplSyntaxOn
\cs_if_exist:NF \__siunitx_print_number:n
  { \msg_critical:nn { typeset } { siunitx-hook } }
\msg_new:nnn { typeset } { siunitx-hook }
  { siunitx~has~no~\__siunitx_print_number:n~to~hook~(numbers~as~written). }
\tl_new:N \l__typeset_number_tl
\cs_set_protected:Npn \__siunitx_print_number:n #1
  {
    \bool_set_false:N \l__siunitx_print_unit_bool
    \bool_if:NTF \l_siunitx_number_parse_bool
      { \use:c { siunitx_print_ \l_siunitx_number_mode_str :n } {#1} }
      {
        \tl_set:Nn \l__typeset_number_tl {#1}
        % (siunitx passes the number already wrapped: \ensuremath{1,500})
        \regex_replace_all:nnN { , } { \cB\{ , \cE\} } \l__typeset_number_tl
        \ensuremath { \symup { \l__typeset_number_tl } }
      }
  }
\ExplSyntaxOff
% XeTeX: tag shaped text with its source characters, so text extraction and
% copy-paste give the characters as written (Indic conjuncts, ligatures).
\XeTeXgenerateactualtext=1

% Bold math under unicode-math (Latin Modern Math has no bold math version,
% so amsmath's \boldsymbol was not bold and \mathbf of Greek had no glyph):
% \boldsymbol and \bm are bold italic like KaTeX's \boldsymbol (digits, which
% have no bold italic form, bold upright); \mathbf is bold upright, Greek
% included. physics' \vb / \vb* use these, so follow. Operators (+, =) have
% no bold glyph in the math font and stay regular.
\ExplSyntaxOn
\tl_new:N \l__typeset_bold_tl
\cs_new_protected:Npn \TypesetBoldSymbol #1
  {
    \tl_set:Nn \l__typeset_bold_tl {#1}
    \regex_replace_all:nnN { [0-9] } { \c{symbfup} \cB\{ \0 \cE\} } \l__typeset_bold_tl
    \symbfit { \l__typeset_bold_tl }
  }
\ExplSyntaxOff
\AtBeginDocument{%
  \renewcommand{\boldsymbol}[1]{\TypesetBoldSymbol{#1}}%
  \renewcommand{\mathbf}[1]{\symbfup{#1}}%
  \providecommand{\bm}{}\renewcommand{\bm}[1]{\TypesetBoldSymbol{#1}}%
}

\def\IndicSetClass#1#2#3{% first, last code point, class
  \IndicChar=#1\relax
  \loop
    \XeTeXcharclass\IndicChar=#3\relax
    \ifnum\IndicChar<#2\relax
    \advance\IndicChar by 1\relax
  \repeat}

% Entering a run opens a group with the script's font; leaving closes it.
\def\IndicTransitions#1#2{%
  \XeTeXinterchartoks 0 #1 = {\begingroup#2}%
  \XeTeXinterchartoks 4095 #1 = {\begingroup#2}%
  \XeTeXinterchartoks #1 0 = {\endgroup}%
  \XeTeXinterchartoks #1 4095 = {\endgroup}}
